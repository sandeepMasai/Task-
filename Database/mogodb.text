Problem 19 — Full Pipeline
db.orders.aggregate([
  {
    $match: {
      status: "delivered"
    }
  },
  {
    $lookup: {
      from: "products",
      localField: "productId",
      foreignField: "_id",
      as: "product"
    }
  },
  {
    $unwind: "$product"
  },
  {
    $addFields: {
      revenueContribution: {
        $divide: ["$amount", 1000]
      }
    }
  },
  {
    $group: {
      _id: "$product.category",
      totalRevenueContribution: {
        $sum: "$revenueContribution"
      },
      orderCount: {
        $sum: 1
      }
    }
  },
  {
    $sort: {
      totalRevenueContribution: -1
    }
  },
  {
    $skip: 1
  },
  {
    $limit: 2
  }
])
